cmake_minimum_required(VERSION 3.24)
project(ice_control_host_tests LANGUAGES C)
enable_testing()
add_executable(test_core test_core.c ../App/app_control.c ../Modules/buttons.c)
target_include_directories(test_core PRIVATE ../App ../Modules)
set_target_properties(test_core PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES)
if(MSVC)
    target_compile_options(test_core PRIVATE /W4 /UNDEBUG)
else()
    target_compile_options(test_core PRIVATE -Wall -Wextra -Werror -UNDEBUG)
endif()
add_test(NAME control_and_buttons COMMAND test_core)
